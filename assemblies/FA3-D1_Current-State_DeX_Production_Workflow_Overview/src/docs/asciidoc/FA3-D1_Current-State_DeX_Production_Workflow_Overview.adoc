<style>
/* Print only (Prince): the screen HTML keeps its normal layout.
   ike-parent's print layout (ike-print.css) puts the generated title alone on page 1 and the
   body after it, with blank pages for double-sided printing. Three of its rules cause that:
   - #content { counter-reset: page 1 }: Prince starts a new page at any page-counter reset,
     so everything after the title page begins on page 2.
   - #header { page-break-after: always; padding-top: 30% }: the title gets a page to itself.
   - body.book #preamble { page-break-before: always } (asciidoctor's own print rules).
   - .book h2 / body.book .sect1>h2 { page-break-before: always }: every chapter-level topic
     starts a new page. Overridden below so chapters continue on the current page; the disclaimers
     keep a page of their own and the body still starts on a fresh page after the contents list.
   Here the title and the cover share page 1, the disclaimers take page 2 and the table of
   contents falls on page 3 (a recto), so there are no blank pages. Page numbering restarts
   at the first chapter instead of at #content. */
@media print {
  html body.book #content { counter-reset: none !important; }
  html body.book #preamble + .sect1 { counter-reset: page 1; }
  html body.book #header { padding-top: 0 !important; page-break-after: auto !important; page: auto !important; }
  html body.book #header > h1:first-child { font-size: 26pt !important; }
  html body.book #header .details { margin-top: 1em !important; }
  html body.book #preamble { page-break-before: auto !important; }
  html body.book .sect1 > h2 { page-break-before: auto !important; }
  html body.book .sect1 > h2#dex-disclaimers { page-break-before: always !important; }
  .cover-page { margin-top: 1.5em; }
  .cover-page .imageblock { margin: 0 0 1em 0; }
  .cover-page .imageblock img { margin-left: auto !important; margin-right: auto !important; }
  .cover-page .imageblock:first-child img { width: 4.4in !important; }
  .cover-page .tableblock { font-family: 'Noto Sans', sans-serif; font-size: 9pt; }
}
</style>
